BUCKET=timecapsule-dev-uploads-leopard
echo $BUCKET

mkdir -p lab4/2026/10 && cd lab4
echo "photo placeholder" > photo.txt
echo "october report" > 2026/10/report.txt
echo "october report 2" > 2026/10/report-2.txt

aws s3 cp photo.txt s3://$BUCKET/uploads/photo.txt
aws s3 sync 2026 s3://$BUCKET/uploads/2026
aws s3 ls s3://$BUCKET --recursive
aws s3 ls s3://$BUCKET
aws s3 ls s3://$BUCKET/uploads/
aws s3api head-object --bucket "$BUCKET" --key "uploads/AssetEditor_art1.png"
aws s3api head-object --bucket "$BUCKET" --key "uploads/2026/10/report.txt"
aws s3 cp s3://$BUCKET/uploads/photo.txt downloaded.txt && cat downloaded.txt
aws s3 rm s3://$BUCKET/uploads/2026/10/report-2.txt

aws s3 ls s3://$BUCKET/uploads/
aws s3 ls s3://$BUCKET/uploads/ --endpoint-url http://s3.eu-central-1.amazonaws.com

echo 'export BUCKET=timecapsule-dev-uploads-leopard' >> ~/.bashrc
source ~/.bashrc
aws s3 presign s3://$BUCKET/uploads/photo.txt --expires-in 120 --region eu-central-1


cd /var/www/mood-diary
ls -la storage/uploads
find . -type d \( -name "uploads" -o -name "attachments" \)
grep -E '^(UPLOAD_DIR|UPLOAD_PATH)=' .env
aws s3 sync storage/uploads s3://$BUCKET/uploads/ --dryrun
aws s3 sync storage/uploads s3://$BUCKET/uploads/
aws s3 sync storage/uploads s3://$BUCKET/uploads/ --dryrun


cd /var/www/mood-diary
git fetch origin
git switch -c feat/cloud-lab4-s3-attachments --track origin/feat/cloud-lab4-s3-attachments
composer update aws/aws-sdk-php --with-all-dependencies

printf '\nS3_BUCKET=timecapsule-dev-uploads-leopard\nAWS_REGION=eu-central-1\n' >> .env

composer update twig/twig --with-all-dependencies
composer show twig/twig
composer audit

git add composer.lock
git commit -m "build: update dependencies for AWS S3 and fix Twig vulnerabilities"
git push origin feat/cloud-lab4-s3-attachments

cd /var/www/mood-diary
git restore composer.lock
git pull --ff-only origin feat/cloud-lab4-s3-attachments
composer install --no-dev --optimize-autoloader

grep -E '^(S3_BUCKET|AWS_REGION)=' .env

aws sts get-caller-identity
aws s3 ls s3://timecapsule-dev-uploads-leopard/uploads/
grep -E '^(S3_BUCKET|AWS_REGION)=' /var/www/mood-diary/.env
